\subsection{SystemVerilog}

\begin{lstlisting}[
	style=engineering,
	language=Verilog,
	caption={Example POR behavioral model.},
	label={lst:por_example}
	]
	module por_vdd (
	input wire porlow,
	output reg por_n
	);
	
	initial begin
	por_n = 1'b0;
	end
	
	always @(posedge porlow) begin
	por_n <= 1'b1;
	end
	
	endmodule
\end{lstlisting}
